function App() {

  const [phase, setPhase] = useState("intro");

  const [nav, setNav] = useState("chat");

  const [woopMsgs, setWoopMsgs] = useState([]);

  const [checkinMsgs, setCheckinMsgs] = useState([]);

  const [typing, setTyping] = useState(false);

  const [woopStep, setWoopStep] = useState(0);

  const [checkinStep, setCheckinStep] = useState(0);

  const [inputVal, setInputVal] = useState("");

  const [data, setData] = useState({wish:"",outcome:"",obstacle:"",plan:""});

  const [woopDone, setWoopDone] = useState(false);

  const bottomRef = useRef(null);


  const addSophieTo = (setter,msg) => setter(m=>[...m,{from:"sophie",...msg,id:Date.now()+Math.random()}]);

  const addUserTo = (setter,text) => setter(m=>[...m,{from:"user",text,id:Date.now()+Math.random()}]);

  const lockLastOf = (setter) => setter(m=>m.map((msg,i)=>i===m.length-1?{...msg,options:null,input:null}:msg));

  const curSetter = () => phase==="woop"?setWoopMsgs:setCheckinMsgs;

  const addSophie = (msg) => addSophieTo(curSetter(),msg);

  const addUser = (text) => addUserTo(curSetter(),text);

  const lockLast = () => lockLastOf(curSetter());

  const withTyping = (fn,delay=1300) => { setTyping(true); setTimeout(()=>{setTyping(false);fn();},delay); };


  const handleNav = (id) => {

    setNav(id);

    if(id==="home") setPhase("home");

    else if(id==="chat"){ if(phase==="home") setPhase(woopDone?"checkin":"woop"); }

  };


  useEffect(()=>{

    if(phase==="woop" && woopMsgs.length===0){

      withTyping(()=>{

        setWoopMsgs([{from:"sophie",id:1,

          text:"Hi Angie! I'm Sophie, your 9amHealth coach. You've been doing really well -- 20 lbs down since you started is something to be proud of.\n\nI want to make sure you feel supported beyond just the medication. Can I walk you through a quick 5-minute exercise that can help healthy habits actually stick long-term?",

          options:["Sure, I'm open to it","What kind of exercise?"]}]);

      },600);

    }

  },[phase]);


  useEffect(()=>{ bottomRef.current?.scrollIntoView({behavior:"smooth"}); },[woopMsgs,checkinMsgs,typing,phase]);


  const handleOption = (opt) => {

    lockLast(); addUser(opt);

    if(phase==="woop") handleWoopOption(opt);

    else handleCheckinOption(opt);

  };


  const handleWoopOption = (opt) => {

    const s = woopStep;

    if(s===0){

      withTyping(()=>{ addSophieTo(setWoopMsgs,{text:"It's called WOOP -- a method backed by behavioral science. We'll clarify what you want, why it matters, and most importantly, plan for the obstacles that usually get in the way.\n\nAt the end, I'll turn your plan into personalized nudges. Ready?",options:["Let's do it"]}); setWoopStep(1); });

    } else if(s===1){

      withTyping(()=>{ addSophieTo(setWoopMsgs,{text:"Think about your health beyond the medication -- your day-to-day lifestyle. What's one thing you wish felt easier or more natural?\n\nIt could be movement, sleep, eating, stress, energy levels... whatever feels most real right now.",input:true,placeholder:"e.g. I wish I could make exercise a consistent habit...",inputKey:"wish"}); setWoopStep(2); });

    } else if(s===4){

      withTyping(()=>{ addSophieTo(setWoopMsgs,{text:"That's a real commitment, Angie. Naming the obstacle is the hardest part -- most people skip it.\n\nI've saved your WOOP and set up your nudges. You'll find them on your dashboard anytime.",special:"woop",woopData:data}); setWoopDone(true); setWoopStep(5); });

    }

  };


  const handleWoopInput = (key) => {

    if(!inputVal.trim()) return;

    const val = inputVal.trim();

    lockLast(); addUser(val);

    const nd = {...data,[key]:val};

    setData(nd); setInputVal("");

    if(key==="wish"){

      withTyping(()=>{ addSophieTo(setWoopMsgs,{text:"That's a really honest one -- and one of the most common things people on this journey want.\n\nNow imagine you actually made this happen. What would be the best outcome? How would you feel day-to-day, or what would be different about your life?",input:true,placeholder:"e.g. I'd have more energy, feel stronger, feel like I'm building something that lasts...",inputKey:"outcome"}); });

    } else if(key==="outcome"){

      withTyping(()=>{ addSophieTo(setWoopMsgs,{text:"That sounds worth building toward.\n\nHere's the important part -- most habits fall apart not because people don't want them, but because something gets in the way. What's the inner obstacle? Not your schedule or logistics, but a feeling or thought that makes it easy to skip.",input:true,placeholder:"e.g. After work I'm just exhausted and can't motivate myself...",inputKey:"obstacle"}); });

    } else if(key==="obstacle"){

      withTyping(()=>{ addSophieTo(setWoopMsgs,{text:"That's such a real one. And honestly, it tracks -- your coach mentioned that your schedule has been really demanding lately, especially with the project you've been managing at work.\n\nLow energy after a full day is one of the most common reasons exercise doesn't stick -- it's not a willpower problem.\n\nLet's make a plan for exactly that moment. Complete this:\n\n\"If I feel too tired to exercise after work, then I will...\"\n\nMake it specific and realistic -- something you could actually do even on a hard day.",input:true,placeholder:"e.g. I'll put on my shoes and do a 20-minute walk, even if that's all I do...",inputKey:"plan"}); });

    } else if(key==="plan"){

      setData(nd);

      withTyping(()=>{ addSophieTo(setWoopMsgs,{text:"You named low energy as your obstacle -- that takes real self-awareness. A lot of people just tell themselves they're lazy.\n\nI also know from your coach that you used to walk your dog in the evenings before things got busy. That might be your built-in plan right there.\n\nBefore I save this: does this feel like a commitment you can actually try this week?",options:["Yes, I'm committing to this","I want to adjust something"]}); setWoopStep(4); });

    }

  };


  const startCheckin = () => {

    setPhase("checkin"); setNav("chat");

    withTyping(()=>{

      setCheckinMsgs([{from:"sophie",id:Date.now(),

        text:"Hey Angie -- it's been a couple of days. I wanted to check in on your plan.\n\nYou committed to: if you feel too tired to exercise after work, you'd put on your shoes and do a 20-minute walk anyway. Your coach also mentioned you were hoping to be more active before your daughter's graduation in June -- so I've been thinking about you. How did it go?",

        options:["I struggled with it","It went pretty well","I forgot about it"]}]);

      setCheckinStep(0);

    },600);

  };


  const handleCheckinOption = (opt) => {

    const s = checkinStep;

    if(s===0){

      withTyping(()=>{ setCheckinMsgs(m=>[...m,{from:"sophie",id:Date.now(),text:"Thank you for being honest -- that's actually the most useful thing you can tell me.\n\nWhat got in the way? Was it the tiredness itself, or something else?",options:["I was exhausted and just couldn't","My schedule got unpredictable","I kept putting it off until it was too late","A mix of all of those"]}]); setCheckinStep(1); });

    } else if(s===1){

      withTyping(()=>{ setCheckinMsgs(m=>[...m,{from:"sophie",id:Date.now(),text:"That makes a lot of sense. An unpredictable schedule is one of the hardest obstacles to plan around -- because by the time the evening arrives, the window is already closed.\n\nYour original plan was built around reclaiming the evening. We need to add a backup for when the day gets away from you.\n\nWant to update your WOOP with that in mind?",options:["Yes, let's update it"]}]); setCheckinStep(2); });

    } else if(s===2){

      withTyping(()=>{ setCheckinMsgs(m=>[...m,{from:"sophie",id:Date.now(),text:"Here's what I'd add to your obstacle: \"...and the unpredictability of my schedule.\"\n\nAnd a backup plan: \"If my evening gets derailed, I'll do a 10-minute walk at lunch instead -- that counts.\"\n\nI'll also add a midday nudge at noon so you have an earlier option before the day takes over.",options:["That feels right, update my plan"]}]); setCheckinStep(3); });

    } else if(s===3){

      withTyping(()=>{

        setCheckinMsgs(m=>[...m,{from:"sophie",id:Date.now(),text:"Done -- your plan is updated.\n\nThe fact that you tried, noticed what got in the way, and came back to refine it? That's exactly how habits form. It's never one clean attempt.\n\nTap home to see your updated plan on your dashboard.",special:"updatedWoop",woopData:data}]);

        setCheckinStep(4);

      });

    }

  };


  const renderMsg = (msg,i,allMsgs) => {

    const isLast = i===allMsgs.length-1;

    const showInteractive = isLast&&(msg.options||msg.input);

    if(msg.from==="user") return (

      <div key={msg.id||i} className="msg-in" style={{display:"flex",gap:8,alignItems:"flex-start",flexDirection:"row-reverse"}}>

        <USER_AVATAR/>

        <div style={{maxWidth:"78%",background:"#ABD4FF",color:"#212121",borderRadius:"14px 3px 14px 14px",padding:"10px 14px",fontSize:14,lineHeight:1.6,whiteSpace:"pre-line"}}>{msg.text}</div>

      </div>

    );

    return (

      <div key={msg.id||i} className="msg-in" style={{display:"flex",gap:8,alignItems:"flex-start"}}>

        <SOPHIE_AVATAR/>

        <div style={{maxWidth:"82%"}}>

          {msg.special ? (

            <>

              <div style={{background:"#F4F4F4",borderRadius:"3px 14px 14px 14px",padding:"10px 14px",fontSize:14,lineHeight:1.6,color:"#212121",whiteSpace:"pre-line"}}>{msg.text}</div>

              <WoopCard wish={msg.woopData?.wish||data.wish} outcome={msg.woopData?.outcome||data.outcome} obstacle={msg.woopData?.obstacle||data.obstacle} plan={msg.woopData?.plan||data.plan} updated={msg.special==="updatedWoop"}/>

              {msg.special==="woop" && isLast && woopDone && (
